<script>
(function () {
  /* ─── 数据 ─── */
  const detailData = JSON.parse(document.getElementById('s72DetailData').textContent);

  /* ─── 工具卡片切换（点缩略图不触发切换）─── */
  document.querySelectorAll('.s72-tool-card').forEach(card => {
    card.addEventListener('click', e => {
      if (e.target.classList.contains('s72-detail-btn')) return;
      if (e.target.closest('[data-lightbox]')) return; /* 点缩略图不切换选中 */
      card.classList.toggle('s72-active');
      syncGiftBox();
    });
  });

  function syncGiftBox() {
    const activeCount = document.querySelectorAll('.s72-tool-card.s72-active').length;
    const giftCard    = document.getElementById('s72GiftCard');
    const hint        = document.getElementById('s72UnlockHint');
    const allSelected = activeCount === 3;
    if (allSelected) {
      giftCard.classList.remove('s72-locked');
      hint.classList.add('s72-hidden');
    } else {
      giftCard.classList.add('s72-locked');
      hint.classList.remove('s72-hidden');
      if (giftCard.classList.contains('s72-active')) {
        giftCard.classList.remove('s72-active');
        document.querySelector('[data-pkg="standard"]').classList.add('s72-active');
      }
    }
  }
  syncGiftBox();

  /* ─── 双列选项互斥 ─── */
  ['pkg', 'ship'].forEach(group => {
    document.querySelectorAll(`[data-${group}]`).forEach(card => {
      card.addEventListener('click', () => {
        if (card.classList.contains('s72-locked')) return;
        document.querySelectorAll(`[data-${group}]`)
          .forEach(c => c.classList.remove('s72-active'));
        card.classList.add('s72-active');
      });
    });
  });

  /* ════════════════════════════════════════
     全屏图片轮播 (Lightbox)
  ════════════════════════════════════════ */
  const lightbox  = document.getElementById('s72Lightbox');
  const lbTrack   = document.getElementById('s72LbTrack');
  const lbDots    = document.getElementById('s72LbDots');
  const lbPrev    = document.getElementById('s72LbPrev');
  const lbNext    = document.getElementById('s72LbNext');
  const lbClose   = document.getElementById('s72LbClose');
  let lbIndex = 0;
  let lbTotal = 0;

  function buildLightbox(images, emoji, startIndex) {
    lbTrack.innerHTML = '';
    lbDots.innerHTML  = '';
    lbIndex = startIndex || 0;
    lbTotal = images.length;

    images.forEach((src, i) => {
      const slide = document.createElement('div');
      slide.className = 's72-lightbox-slide';
      if (src) {
        slide.innerHTML = `<img src="${src}" alt="产品图 ${i+1}">`;
      } else {
        slide.innerHTML = `<div class="s72-lb-placeholder">${emoji}</div>`;
      }
      lbTrack.appendChild(slide);

      const dot = document.createElement('div');
      dot.className = 's72-lb-dot' + (i === lbIndex ? ' s72-active' : '');
      lbDots.appendChild(dot);
    });

    updateLbPosition(false);
    lightbox.classList.add('s72-open');
    document.body.style.overflow = 'hidden';
  }

  function updateLbPosition(animate) {
    lbTrack.style.transition = animate === false ? 'none' : 'transform 0.36s cubic-bezier(0.32,0.72,0,1)';
    lbTrack.style.transform  = `translateX(-${lbIndex * 100}%)`;
    document.querySelectorAll('.s72-lb-dot').forEach((d, i) => {
      d.classList.toggle('s72-active', i === lbIndex);
    });
    lbPrev.disabled = lbIndex === 0;
    lbNext.disabled = lbIndex === lbTotal - 1;
  }

  function closeLightbox() {
    lightbox.classList.remove('s72-open');
    document.body.style.overflow = '';
  }

  lbPrev.addEventListener('click', () => { if (lbIndex > 0) { lbIndex--; updateLbPosition(); } });
  lbNext.addEventListener('click', () => { if (lbIndex < lbTotal - 1) { lbIndex++; updateLbPosition(); } });
  lbClose.addEventListener('click', closeLightbox);
  document.getElementById('s72LbBackBtn').addEventListener('click', closeLightbox);
  lightbox.addEventListener('click', e => { if (e.target === lightbox) closeLightbox(); });

  /* 触摸滑动轮播 */
  let lbTouchX = 0;
  lbTrack.addEventListener('touchstart', e => { lbTouchX = e.touches[0].clientX; }, { passive: true });
  lbTrack.addEventListener('touchend', e => {
    const dx = e.changedTouches[0].clientX - lbTouchX;
    if (dx < -50 && lbIndex < lbTotal - 1) { lbIndex++; updateLbPosition(); }
    if (dx >  50 && lbIndex > 0)           { lbIndex--; updateLbPosition(); }
  }, { passive: true });

  /* 键盘支持 */
  document.addEventListener('keydown', e => {
    if (!lightbox.classList.contains('s72-open')) return;
    if (e.key === 'ArrowLeft'  && lbIndex > 0)            { lbIndex--; updateLbPosition(); }
    if (e.key === 'ArrowRight' && lbIndex < lbTotal - 1)  { lbIndex++; updateLbPosition(); }
    if (e.key === 'Escape') closeLightbox();
  });

  /* 缩略图点击 → 打开轮播 */
  const emojiMap = { pliers: '🔧', wrench: '🔩', axe: '🪓' };
  document.querySelectorAll('[data-lightbox]').forEach(thumb => {
    thumb.addEventListener('click', e => {
      e.stopPropagation();
      const key    = thumb.dataset.lightbox;
      const d      = detailData[key];
      if (!d) return;
      /* 用详情数据里的 images；无图则显示 3 张 emoji 占位 */
      const imgs   = (d.images && d.images.length) ? d.images : [null, null, null];
      buildLightbox(imgs, emojiMap[key] || '📦', 0);
    });
  });

  /* ════════════════════════════════════════
     抽屉
  ════════════════════════════════════════ */
  const overlay    = document.getElementById('s72Overlay');
  const drawer     = document.getElementById('s72Drawer');
  const drawerTitle= document.getElementById('s72DrawerTitle');
  const drawerBody = document.getElementById('s72DrawerBody');
  const backBtn    = document.getElementById('s72DrawerBack');
  const floatBack  = document.getElementById('s72FloatBack');

  function openDrawer(key) {
    const d = detailData[key];
    if (!d) return;

    drawerTitle.textContent = d.title;

    /* 视频区 */
    let videoHTML = '';
    if (d.videoUrl) {
      const isYT = d.videoUrl.includes('youtu');
      videoHTML = isYT
        ? `<div class="s72-detail-video"><iframe src="${d.videoUrl}" allowfullscreen></iframe></div>`
        : `<div class="s72-detail-video"><video src="${d.videoUrl}" controls playsinline></video></div>`;
    } else {
      videoHTML = `<div class="s72-video-placeholder">
        <svg width="40" height="40" viewBox="0 0 24 24" fill="none">
          <circle cx="12" cy="12" r="10" stroke="#555" stroke-width="1.5"/>
          <polygon points="10,8 18,12 10,16" fill="#fff"/>
        </svg>
        <span>演示视频即将上线</span>
      </div>`;
    }

    /* 图片区（网格，可点击放大） */
    let imgHTML = '<div class="s72-detail-images">';
    if (d.images && d.images.length) {
      d.images.forEach((url, i) => {
        imgHTML += `<div class="s72-detail-img-slot" style="cursor:zoom-in" data-lb-key="${key}" data-lb-idx="${i}">
          <img src="${url}" loading="lazy"></div>`;
      });
    } else {
      for (let i = 0; i < 4; i++) {
        imgHTML += `<div class="s72-detail-img-slot">产品图 ${i+1}</div>`;
      }
    }
    imgHTML += '</div>';

    /* 卖点 */
    const featHTML = '<ul class="s72-feature-list">'
      + d.features.map(f => `<li>${f}</li>`).join('')
      + '</ul>';

    /* 规格 */
    const specHTML = '<table class="s72-spec-table">'
      + d.specs.map(([k,v]) => `<tr><td>${k}</td><td>${v}</td></tr>`).join('')
      + '</table>';

    drawerBody.innerHTML = `
      ${videoHTML}
      ${imgHTML}
      <div class="s72-detail-section-title">核心卖点</div>
      ${featHTML}
      <div class="s72-detail-section-title">产品参数</div>
      ${specHTML}
    `;
    drawerBody.scrollTop = 0;

    /* 抽屉内图片也可点击放大 */
    drawerBody.querySelectorAll('[data-lb-key]').forEach(slot => {
      slot.addEventListener('click', () => {
        const k   = slot.dataset.lbKey;
        const idx = parseInt(slot.dataset.lbIdx, 10);
        const imgs = detailData[k].images;
        buildLightbox(imgs, emojiMap[k] || '📦', idx);
      });
    });

    overlay.classList.add('s72-open');
    drawer.classList.add('s72-open');
    document.body.style.overflow = 'hidden';
  }

  function closeDrawer() {
    overlay.classList.remove('s72-open');
    drawer.classList.remove('s72-open');
    document.body.style.overflow = '';
  }

  document.querySelectorAll('.s72-detail-btn').forEach(btn => {
    btn.addEventListener('click', e => {
      e.stopPropagation();
      openDrawer(btn.dataset.detail);
    });
  });

  backBtn.addEventListener('click', closeDrawer);
  floatBack.addEventListener('click', closeDrawer);
  overlay.addEventListener('click', closeDrawer);

  /* 手势下滑关闭抽屉 */
  let startY = 0;
  drawer.addEventListener('touchstart', e => { startY = e.touches[0].clientY; }, { passive: true });
  drawer.addEventListener('touchend', e => {
    if (e.changedTouches[0].clientY - startY > 60) closeDrawer();
  }, { passive: true });

  /* Shopify 动态 Section 重载支持 */
  document.addEventListener('shopify:section:load', () => { syncGiftBox(); });
})();
</script>
